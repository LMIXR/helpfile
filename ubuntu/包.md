# 查看依赖
1. `apt-cache depends XXX`

# 下载deb及其依赖包
1. `apt-get download $(apt-cache depends --recurse --no-recommends --no-suggests --no-conflicts --no-breaks --no-replaces --no-enhances --no-pre-depends XXX | grep -v amd64 | grep "^\w")`

# 使用dpkg命令离线安装
1. sudo dpkg -i *.deb

# 离线下载deb包

1. `mkdir openssh`, `cd openssh`
2. `apt-rdepends openssh-server | grep "^\w" > openssh.txt`
3. `for pkg in $(cat .openssh.txt); do apt-get download $pkg; done`
